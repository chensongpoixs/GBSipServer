void SipServer::HandlerSipMessageAnswerd(eXosip_event_t * sip_event) {
    if (!sip_event || !sip_event->request) {
        return;
    }
    
    // 获取消息体
    osip_body_t* body = nullptr;
    osip_message_get_body(sip_event->request, 0, &body);
    if (!body || !body->body) {
        SIPSERVER_LOG(LS_WARNING) << "MESSAGE response has no body";
        return;
    }
    
    std::string xml(body->body);
    
    // 解析CmdType
    auto root = gbsip_server::XmlHelper::parseXml(xml);
    if (!root) {
        SIPSERVER_LOG(LS_WARNING) << "Failed to parse XML";
        return;
    }
    
    std::string cmdType = gbsip_server::XmlHelper::getNodeValue(root, "Response/CmdType");
    
    // 分发到对应的Handler
    if (cmdType == "Catalog") {
        if (catalog_handler_) {
            catalog_handler_->handleMessageResponse(sip_event);
        }
    } else if (cmdType == "DeviceInfo") {
        // TODO: 处理设备信息响应
    } else if (cmdType == "DeviceStatus") {
        // TODO: 处理设备状态响应
    } else {
        SIPSERVER_LOG(LS_WARNING) << "Unknown CmdType: " << cmdType;
    }
}

### 2.4 添加 HandlerSipInSubscriptionNew 方法处理NOTIFY
```cpp
void SipServer::HandlerSipInSubscriptionNew(eXosip_event_t * sip_event) {
    if (!sip_event || !sip_event->request) {
        return;
    }
    
    // 获取Event头
    osip_header_t* event_header = nullptr;
    osip_message_get_header(sip_event->request, 0, &event_header);
    if (!event_header || !event_header->hvalue) {
        SIPSERVER_LOG(LS_WARNING) << "NOTIFY has no Event header";
        return;
    }
    
    std::string event(event_header->hvalue);
    
    // 分发到对应的Handler
    if (event == "Catalog") {
        if (catalog_handler_) {
            catalog_handler_->handleNotify(sip_event);
        }
    }
    
    // 发送200 OK响应
    eXosip_lock(sip_context_);
    eXosip_call_send_answer(sip_context_, sip_event->tid, 200, nullptr);
    eXosip_unlock(sip_context_);
}
```

### 3.1 修改 queryCatalog 方法
```cpp
#include "sip_server.h"
#include <future>

oatpp::Object<CatalogResponseDto> CatalogService::queryCatalog(
    const oatpp::String& deviceId) {
    
    // 检查设备是否存在
    auto device = m_deviceDb->getDeviceByDeviceId(deviceId);
    if (!device) {
        return nullptr;
    }
    
    // 获取SipServer实例
    auto& sipServer = gbsip_server::SipServer::GetInstance();
    auto catalogHandler = sipServer.getCatalogHandler();
    if (!catalogHandler) {
        return nullptr;
    }
    
    // 获取设备信息
    auto deviceInfo = sipServer.getDeviceInfo(deviceId->c_str());
    if (!deviceInfo) {
        return nullptr;
    }
    
    // 创建Promise用于同步等待
    auto promise = std::make_shared<std::promise<gbsip_server::CatalogResponse>>();
    auto future = promise->get_future();
    
    // 调用CatalogHandler查询
    catalogHandler->queryCatalog(
        deviceId->c_str(),
        deviceInfo->ip,
        deviceInfo->port,
        [promise](const gbsip_server::CatalogResponse& response) {
            promise->set_value(response);
        },
        [promise](int code, const std::string& message) {
            promise->set_exception(
                std::make_exception_ptr(std::runtime_error(message))
            );
        }
    );
    
    // 等待响应（超时30秒）
    auto status = future.wait_for(std::chrono::seconds(30));
    if (status == std::future_status::timeout) {
        return nullptr;
    }
    
    try {
        auto response = future.get();
        
        // 保存到数据库
        for (const auto& channelInfo : response.channels) {
            auto channel = ChannelDto::createShared();
            channel->channelId = channelInfo.channelId;
            channel->deviceId = deviceId;
            channel->name = channelInfo.name;
            channel->manufacturer = channelInfo.manufacturer;
            channel->model = channelInfo.model;
            channel->owner = channelInfo.owner;
            channel->civilCode = channelInfo.civilCode;
            channel->address = channelInfo.address;
            channel->parental = channelInfo.parental;
            channel->parentId = channelInfo.parentId;
            channel->safetyWay = channelInfo.safetyWay;
            channel->registerWay = channelInfo.registerWay;
            channel->secrecy = channelInfo.secrecy;
            channel->status = channelInfo.status;
            channel->longitude = channelInfo.longitude;
            channel->latitude = channelInfo.latitude;
            channel->ptzType = channelInfo.ptzType;
            channel->updateTime = getCurrentTimestamp();
            channel->createTime = channel->updateTime;
            
            saveOrUpdateChannel(channel);
        }
        
        // 构建响应DTO
        auto dto = CatalogResponseDto::createShared();
        dto->deviceId = deviceId;
        dto->sn = std::stoi(response.sn);
        dto->sumNum = response.sumNum;
        dto->channels = oatpp::Vector<oatpp::Object<ChannelDto>>::createShared();
        
        // 从数据库读取通道列表（确保数据一致性）
        auto channels = getChannelsByDeviceId(deviceId);
        dto->channels = channels;
        
        return dto;
    } catch (const std::exception& e) {
        SIPSERVER_LOG(LS_ERROR) << "Query catalog failed: " << e.what();
        return nullptr;
    }
}
```

### 3.2 修改 subscribeCatalog 方法
```cpp
oatpp::Object<StatusDto> CatalogService::subscribeCatalog(
    const oatpp::Object<CatalogSubscribeDto>& dto) {
    
    auto status = StatusDto::createShared();
    
    // 检查设备是否存在
    auto device = m_deviceDb->getDeviceByDeviceId(dto->deviceId);
    if (!device) {
        status->code = 404;
        status->status = "ERROR";
        status->message = "Device not found";
        return status;
    }
    
    // 获取SipServer实例
    auto& sipServer = gbsip_server::SipServer::GetInstance();
    auto catalogHandler = sipServer.getCatalogHandler();
    if (!catalogHandler) {
        status->code = 500;
        status->status = "ERROR";
        status->message = "Catalog handler not initialized";
        return status;
    }
    
    // 获取设备信息
    auto deviceInfo = sipServer.getDeviceInfo(dto->deviceId->c_str());
    if (!deviceInfo) {
        status->code = 404;
        status->status = "ERROR";
        status->message = "Device not online";
        return status;
    }
    
    // 创建Promise用于同步等待
    auto promise = std::make_shared<std::promise<void>>();
    auto future = promise->get_future();
    
    // 调用CatalogHandler订阅
    catalogHandler->subscribeCatalog(
        dto->deviceId->c_str(),
        deviceInfo->ip,
        deviceInfo->port,
        dto->expires,
        [promise]() {
            promise->set_value();
        },
        [promise](int code, const std::string& message) {
            promise->set_exception(
                std::make_exception_ptr(std::runtime_error(message))
            );
        }
    );
    
    // 等待响应（超时10秒）
    auto waitStatus = future.wait_for(std::chrono::seconds(10));
    if (waitStatus == std::future_status::timeout) {
        status->code = 408;
        status->status = "ERROR";
        status->message = "Subscribe timeout";
        return status;
    }
    
    try {
        future.get();
        status->code = 200;
        status->status = "OK";
        status->message = "Catalog subscribe success";
    } catch (const std::exception& e) {
        status->code = 500;
        status->status = "ERROR";
        status->message = e.what();
    }
    
    return status;
}
```
