```cpp
// service/AlarmService.cpp

oatpp::Object<StatusDto> AlarmService::confirmAlarm(
    const oatpp::Object<AlarmConfirmDto>& dto) {
    
    auto response = StatusDto::createShared();
    
    try {
        // 验证报警是否存在
        auto alarm = m_alarmDb->getAlarmById(dto->alarm_id);
        if (!alarm) {
            response->code = 404;
            response->message = "Alarm not found";
            return response;
        }
        
        // 确认报警
        bool success = m_alarmDb->confirmAlarm(
            dto->alarm_id,
            dto->operator_name,
            dto->remark
        );
        
        if (success) {
            response->code = 200;
            response->message = "Alarm confirmed successfully";
            
            // 触发WebSocket推送（如果已实现）
            notifyAlarmUpdate(alarm);
        } else {
            response->code = 500;
            response->message = "Failed to confirm alarm";
        }
        
    } catch (const std::exception& e) {
        OATPP_LOGE("AlarmService", "confirmAlarm error: %s", e.what());
        response->code = 500;
        response->message = e.what();
    }
    
    return response;
}

oatpp::Object<StatusDto> AlarmService::resetAlarm(
    const oatpp::Object<AlarmResetDto>& dto) {
    
    auto response = StatusDto::createShared();
    
    try {
        // 发送报警复位MESSAGE到设备
        std::string xml = buildAlarmResetXml(dto->device_id, dto->alarm_type);
        
        // 获取设备信息
        auto device = m_deviceDb->getDeviceById(dto->device_id);
        if (!device) {
            response->code = 404;
            response->message = "Device not found";
            return response;
        }
        
        // 发送SIP MESSAGE
        bool sent = m_alarmHandler->sendAlarmReset(
            dto->device_id,
            device->ip,
            device->port,
            xml
        );
        
        if (sent) {
            // 更新数据库
            m_alarmDb->resetAlarm(dto->device_id, dto->alarm_type);
            
            response->code = 200;
            response->message = "Alarm reset successfully";
        } else {
            response->code = 500;
            response->message = "Failed to send reset command";
        }
        
    } catch (const std::exception& e) {
        OATPP_LOGE("AlarmService", "resetAlarm error: %s", e.what());
        response->code = 500;
        response->message = e.what();
    }
    
    return response;
}

std::string AlarmService::buildAlarmResetXml(
    const std::string& deviceId,
    const std::string& alarmType) {
    
    std::ostringstream oss;
    oss << "<?xml version=\"1.0\" encoding=\"GB2312\"?>\n"
        << "<Control>\n"
        << "  <CmdType>AlarmReset</CmdType>\n"
        << "  <SN>" << generateSN() << "</SN>\n"
        << "  <DeviceID>" << deviceId << "</DeviceID>\n"
        << "  <AlarmMethod>" << alarmType << "</AlarmMethod>\n"
        << "</Control>";
    
    return oss.str();
}
```

```cpp
namespace gbsip_server {

// G.711 A-law编码实现
std::vector<uint8_t> AudioCodec::encodeG711A(const std::vector<int16_t>& pcm) {
    std::vector<uint8_t> result;
    result.reserve(pcm.size());
    
    for (int16_t sample : pcm) {
        result.push_back(encodeAlawSample(sample));
    }
    
    return result;
}

uint8_t AudioCodec::encodeAlawSample(int16_t sample) {
    const uint16_t ALAW_MAX = 0xFFF;
    uint8_t sign = (sample < 0) ? 0x80 : 0x00;
    
    if (sign) {
        sample = -sample - 1;
    }
    
    if (sample > ALAW_MAX) {
        sample = ALAW_MAX;
    }
    
    // 查找段位
    uint8_t segment = 0;
    uint16_t temp = sample + 8;
    
    if (temp >= 256) {
        segment = 1;
        while (temp >= 256) {
            temp >>= 1;
            segment++;
        }
    }
    
    // 计算量化值
    uint8_t quantization = (temp >> 4) & 0x0F;
    
    // 组合结果
    uint8_t alaw = sign | (segment << 4) | quantization;
    
    // A-law异或0x55
    return alaw ^ 0x55;
}

std::vector<int16_t> AudioCodec::decodeG711A(const std::vector<uint8_t>& g711) {
    std::vector<int16_t> result;
    result.reserve(g711.size());
    
    for (uint8_t encoded : g711) {
        // A-law异或0x55
        encoded ^= 0x55;
        
        uint8_t sign = encoded & 0x80;
        uint8_t segment = (encoded >> 4) & 0x07;
        uint8_t quantization = encoded & 0x0F;
        
        // 计算PCM值
        int16_t sample = (quantization << 4) + 8;
        
        if (segment > 0) {
            sample += 0x100;
            sample <<= (segment - 1);
        }
        
        if (sign) {
            sample = -sample;
        }
        
        result.push_back(sample);
    }
    
    return result;
}
```

```cpp
// sip/AudioRTPHandler.hpp
#ifndef GBSIPSERVER_AUDIORTPHANDLER_HPP
#define GBSIPSERVER_AUDIORTPHANDLER_HPP

#include <cstdint>
#include <vector>
#include <string>

namespace gbsip_server {

// RTP头结构
struct RTPHeader {
    uint8_t version:2;      // 版本（固定为2）
    uint8_t padding:1;      // 填充标志
    uint8_t extension:1;    // 扩展标志
    uint8_t csrc_count:4;   // CSRC计数
    uint8_t marker:1;       // 标记位
    uint8_t payload_type:7; // 负载类型
    uint16_t sequence;      // 序列号
    uint32_t timestamp;     // 时间戳
    uint32_t ssrc;          // 同步源标识符
};

class AudioRTPHandler {
public:
    AudioRTPHandler();
    ~AudioRTPHandler();
    
    /**
     * 封装音频RTP包
     * @param audio 音频数据
     * @param timestamp RTP时间戳
     * @param marker 标记位
     * @return RTP包数据
     */
    std::vector<uint8_t> packAudioRTP(
        const std::vector<uint8_t>& audio,
        uint32_t timestamp,
        bool marker = false
    );
    
    /**
     * 解析音频RTP包
     * @param data RTP包数据
     * @param audio 输出音频数据
     * @param timestamp 输出时间戳
     * @return 是否解析成功
     */
    bool unpackAudioRTP(
        const uint8_t* data,
        size_t len,
        std::vector<uint8_t>& audio,
        uint32_t& timestamp
    );
    
    /**
     * 设置SSRC
     */
    void setSSRC(uint32_t ssrc) { m_ssrc = ssrc; }
    
    /**
     * 设置负载类型
     */
    void setPayloadType(uint8_t pt) { m_payloadType = pt; }
    
private:
    uint32_t m_ssrc;           // SSRC
    uint16_t m_sequence;       // 序列号
    uint8_t m_payloadType;     // 负载类型（8=G.711A, 0=G.711U）
};

} // namespace gbsip_server

#endif // GBSIPSERVER_AUDIORTPHANDLER_HPP
```

#### 2.4.2 实现AudioRTPHandler

```cpp
// sip/AudioRTPHandler.cpp
#include "AudioRTPHandler.hpp"
#include <cstring>
#include <arpa/inet.h>

namespace gbsip_server {

AudioRTPHandler::AudioRTPHandler()
    : m_ssrc(0)
    , m_sequence(0)
    , m_payloadType(8) // 默认G.711A
{
}

AudioRTPHandler::~AudioRTPHandler() {
}

std::vector<uint8_t> AudioRTPHandler::packAudioRTP(
    const std::vector<uint8_t>& audio,
    uint32_t timestamp,
    bool marker)
{
    std::vector<uint8_t> packet;
    packet.resize(sizeof(RTPHeader) + audio.size());
    
    // 填充RTP头
    RTPHeader* header = reinterpret_cast<RTPHeader*>(packet.data());
    header->version = 2;
    header->padding = 0;
    header->extension = 0;
    header->csrc_count = 0;
    header->marker = marker ? 1 : 0;
    header->payload_type = m_payloadType;
    header->sequence = htons(m_sequence++);
    header->timestamp = htonl(timestamp);
    header->ssrc = htonl(m_ssrc);
    
    // 复制音频数据
    std::memcpy(packet.data() + sizeof(RTPHeader), 
                audio.data(), 
                audio.size());
    
    return packet;
}

bool AudioRTPHandler::unpackAudioRTP(
    const uint8_t* data,
    size_t len,
    std::vector<uint8_t>& audio,
    uint32_t& timestamp)
{
    if (len < sizeof(RTPHeader)) {
        return false;
    }
    
    const RTPHeader* header = reinterpret_cast<const RTPHeader*>(data);
    
    // 验证版本
    if (header->version != 2) {
        return false;
    }
    
    // 提取时间戳
    timestamp = ntohl(header->timestamp);
    
    // 提取音频数据
    size_t audioLen = len - sizeof(RTPHeader);
    audio.resize(audioLen);
    std::memcpy(audio.data(), 
                data + sizeof(RTPHeader), 
                audioLen);
    
    return true;
}

} // namespace gbsip_server
```
